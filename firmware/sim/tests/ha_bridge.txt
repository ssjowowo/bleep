# The remote in Home Assistant: its battery as entities, key presses as events
click Start
wait 300
click Home
wait 2000
expect-log ha: state sensor.bleep_7c2f1a_battery = 82
expect-log ha: state binary_sensor.bleep_7c2f1a_charging = off
# a change is reported again; no change, nothing
mark
battery 64
wait 1500
expect-log-count 1 ha: state sensor.bleep_7c2f1a_battery = 64
wait 3000
expect-log-count 1 ha: state sensor.bleep_7c2f1a_battery
# key events are off by default
mark
key MUTE
wait 300
expect-log-no fire_event
# turn them on (Settings > Home Assistant)
tap 280 450
wait 300
click Home Assistant
wait 600
expect THIS REMOTE IN HOME ASSISTANT
