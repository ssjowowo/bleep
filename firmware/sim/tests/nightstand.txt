# Nightstand: idle on USB, a big clock, the date and Home Assistant's weather
click Start
wait 300
usb on
wait 300
expect Charging · USB-C
# idle: the dim screen is the nightstand clock; Wi-Fi stays on for the weather
wait 9000
expect 21:4
expect Thursday 8 October
expect 14° · Cloudy · H 17° · L 9°
expect Charging · 82%
expect-log wifi: connected
shot 83_nightstand
# the lights go out: the clock drops to the lowest backlight, and back up with them
mark
lux 2
wait 1500
expect-log power: dim backlight 2%
lux 400
wait 1500
expect-log power: dim backlight 17%
# it stays up all night: no screen off while on USB
wait 120000
expect Thursday 8 October
# a tap wakes the remote; unplugged, it's the normal lock screen when idle
tap 50% 50%
wait 300
expect-no Thursday 8 October
usb off
wait 9500
expect Pick up or press any key
expect-no 14° · Cloudy
# the setting (OK with nothing to control opens the keys sheet: BACK closes it)
key OK
wait 300
key BACK
wait 300
tab Settings
wait 300
click Battery & sleep
wait 300
expect Nightstand clock
